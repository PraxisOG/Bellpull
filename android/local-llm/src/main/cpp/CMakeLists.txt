cmake_minimum_required(VERSION 3.31.6)
project(bellpull_llm LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
add_compile_options(-O3)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../" ABSOLUTE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
if(ANDROID_ABI STREQUAL "arm64-v8a")
    set(GGML_CPU_KLEIDIAI ON CACHE BOOL "" FORCE)
    set(GGML_VULKAN ON CACHE BOOL "" FORCE)
    set(SPIRV-Headers_DIR "${CMAKE_CURRENT_SOURCE_DIR}/cmake" CACHE PATH "" FORCE)
    set(Vulkan_GLSLC_EXECUTABLE "${CMAKE_ANDROID_NDK}/shader-tools/linux-x86_64/glslc" CACHE FILEPATH "" FORCE)
    find_program(BELLPULL_HOST_CC NAMES cc gcc clang REQUIRED NO_CMAKE_FIND_ROOT_PATH)
    find_program(BELLPULL_HOST_CXX NAMES c++ g++ clang++ REQUIRED NO_CMAKE_FIND_ROOT_PATH)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/shader-host.cmake"
        "set(CMAKE_C_COMPILER \"${BELLPULL_HOST_CC}\")\nset(CMAKE_CXX_COMPILER \"${BELLPULL_HOST_CXX}\")\nset(CMAKE_MAKE_PROGRAM \"${CMAKE_MAKE_PROGRAM}\" CACHE FILEPATH \"\" FORCE)\nset(CMAKE_FIND_ROOT_PATH_MODE_PROGRAM NEVER)\nset(CMAKE_FIND_ROOT_PATH_MODE_LIBRARY NEVER)\nset(CMAKE_FIND_ROOT_PATH_MODE_INCLUDE NEVER)\n")
    set(GGML_VULKAN_SHADERS_GEN_TOOLCHAIN "${CMAKE_CURRENT_BINARY_DIR}/shader-host.cmake" CACHE FILEPATH "" FORCE)
    include_directories("${CMAKE_ANDROID_NDK}/sources/third_party/shaderc/third_party/spirv-tools/external/spirv-headers/include" "${ROOT}/third-party/Vulkan-Headers/include")
else()
    set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
    set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
endif()
add_subdirectory("${ROOT}/third-party/llama.cpp" llama EXCLUDE_FROM_ALL)
add_library(bellpull_llm SHARED local-llm.cpp)
target_include_directories(bellpull_llm PRIVATE "${ROOT}/third-party/llama.cpp/common")
target_link_libraries(bellpull_llm PRIVATE llama llama-common android log)
# Qwen speech embeds another GGML revision. Never export/interpose either runtime.
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/exports.map" "{ global: Java_org_praxisog_hermes_local_LocalLlmNative_*; local: *; };\n")
target_link_options(bellpull_llm PRIVATE "-Wl,--exclude-libs,ALL" "-Wl,-z,max-page-size=16384" "-Wl,--version-script=${CMAKE_CURRENT_BINARY_DIR}/exports.map")
